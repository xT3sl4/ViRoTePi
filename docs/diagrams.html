ViRoTePi - Diagramy UML

System zarzadzania paczkami typu InPost

Legenda: Diagram przedstawia encje bazodanowe (backend, MariaDB) oraz klasy okien WPF (frontend). Linie ciagle = relacje bazodanowe (FK). Linie przerywane = nawigacja miedzy oknami.
Diagram klas - Encje bazodanowe
classDiagram
    direction LR

    class user {
        +int id
        +string name
        +string surname
        +string email
        +DateOnly date_of_birth
        +string password
        +string role
        +int phone_number
        +string selfie
        +bool sex
    }

    class admin {
        +int admin_id
        +int user_id
    }

    class klient {
        +int klient_id
        +int user_id
        +int pack_id
    }

    class kurier {
        +int kurier_id
        +int user_id
        +string state
    }

    class pack {
        +int pack_id
        +string size
        +int klient_id
        +bool delivered
        +DateOnly date
        +DateOnly to_when
        +bool picked_up
    }

    class pending_pack {
        +int pending_id
        +int sender_klient_id
        +int receiver_klient_id
        +string size
        +int paczkomat_id
        +DateTime created_at
        +string status
    }

    class box {
        +int box_id
        +int pack_id
        +string size
    }

    class paczkomat {
        +int paczkomat_id
        +string paczkomat_name
        +string address
    }

    class kuriers_datum {
        +int kurier_id
        +int pack_id
    }

    class paczkomat_datum {
        +int paczkomat_id
        +int box_id
    }

    user "1" --> "*" admin : has
    user "1" --> "*" klient : has
    user "1" --> "*" kurier : has

    klient "1" --> "*" pack : owns
    klient "1" --> "*" pending_pack : sends
    klient "1" --> "*" pending_pack : receives

    pack "1" --> "*" box : contains
    box "1" --> "0..1" paczkomat_datum : stored in
    paczkomat "1" --> "*" paczkomat_datum : holds

    kurier "1" --> "*" kuriers_datum : assigned
    pack "1" --> "*" kuriers_datum : delivered by

    pending_pack "*" --> "1" paczkomat : target
